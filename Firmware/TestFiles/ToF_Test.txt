#include <Arduino.h>
#include <Wire.h>
#include <Adafruit_VL53L0X.h>

// pin assignments

constexpr int I2C_SDA_PIN = 7;
constexpr int I2C_SCL_PIN = 15;

constexpr int TOF_XSHUT_PIN = 14;

constexpr uint8_t TOF_I2C_ADDRESS = 0x29;

// create an instance of the VL53L0X sensor object

Adafruit_VL53L0X tof;

// setup

void setup() {

  Serial.begin(115200);
  delay(1000);

  Serial.println();
  Serial.println("DriveWire VL53L0X Test Starting");

  // set XSHUT as an output
  pinMode(TOF_XSHUT_PIN, OUTPUT);

  // toggle XSHUT to reset sensor 
  digitalWrite(TOF_XSHUT_PIN, LOW);
  delay(10);

  digitalWrite(TOF_XSHUT_PIN, HIGH); 
  delay(10);

  Serial.println("VL53L0X hardware reset complete"); 


  // initialize I2C bus communication

  Wire.begin(I2C_SDA_PIN, I2C_SCL_PIN);
  Serial.println("I2C bus initialized");

  // initialize the sensor

  if (!tof.begin(TOF_I2C_ADDRESS, false, &Wire)) {

    Serial.println("Failed to initialize sensor");

    while (true) {
      delay(100);
    }
  }

  Serial.println("Sensor successfully initialized");
  Serial.println("Beginning distance measurements");
  Serial.println();

}

// loop
void loop() {
VL53L0X_RangingMeasurementData_t measurement; 

// request a measurement
tof.rangingTest(&measurement, false); // address of data to fill, debug flag

if (measurement.RangeStatus != 4) { // if not out of range
  Serial.print("Distance (mm): ");
  Serial.print(measurement.RangeMilliMeter);
  Serial.print("mm | ");
  Serial.print(measurement.RangeMilliMeter / 10.0);
  Serial.println(" cm");
} else {
  Serial.println("Out of range");
}

delay(1000); // wait 1 second before next measurement
}
